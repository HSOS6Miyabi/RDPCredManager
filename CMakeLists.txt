cmake_minimum_required(VERSION 3.20)
project(RDPCredManager LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "RDPCredManager requires 64-bit Windows")
endif()
add_executable(RDPCredManager WIN32 src/main.cpp)
target_compile_features(RDPCredManager PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(RDPCredManager PRIVATE /utf-8 /Oi- /GS- /GR-)
    target_link_options(RDPCredManager PRIVATE /ENTRY:WinMainCRTStartup /NODEFAULTLIB
        /MANIFEST:EMBED /MANIFESTUAC:NO
        /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
endif()
target_link_libraries(RDPCredManager PRIVATE user32 gdi32 advapi32 shell32 kernel32 comctl32 uxtheme)
add_executable(manifest_verify tests/manifest_verify.cpp)
add_executable(core_test tests/core_test.cpp)
add_executable(layout_test tests/layout_test.cpp)
add_executable(win_credential_test tests/win_credential_test.cpp)
add_executable(win_gui_test tests/win_gui_test.cpp)
target_link_libraries(win_gui_test PRIVATE user32 advapi32)
target_link_libraries(win_credential_test PRIVATE advapi32)
enable_testing()
add_test(NAME embedded_manifest COMMAND manifest_verify $<TARGET_FILE:RDPCredManager>)
add_test(NAME core COMMAND core_test)
add_test(NAME layout COMMAND layout_test)
add_test(NAME credential_manager_roundtrip COMMAND win_credential_test)
add_test(NAME win_gui_save_edit COMMAND win_gui_test)
